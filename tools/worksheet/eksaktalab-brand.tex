\newcommand{\EkIconAt}[2]{%
  \begin{scope}[shift={(#1)}]
    \fill[EkTeal, rounded corners=0.2*#2] (-0.5*#2,-0.5*#2) rectangle (0.5*#2,0.5*#2);
    \begin{scope}[shift={(-0.5*#2,0.5*#2)}, x=#2/448, y=-#2/448,
                  white, line width=0.085*#2, line cap=round, line join=round]
      \draw (128,168) -- (208,168);
      \draw (128,224) -- (228,224);
      \draw (128,280) -- (208,280);
      \draw (258,164) -- (318,224) -- (258,284);
    \end{scope}
  \end{scope}%
}

% --- Judul sub-materi (standar semua worksheet), mis. \subtopik{A.1}{Barisan Aritmatika}.
%     Penomoran mengikuti kode latihan online (EKS-A1, BD-A1, ...). ---
\providecommand{\subtopik}[2]{%
  \par\noindent\begin{tikzpicture}
    \fill[MatnAccent] (0,0) rectangle (0.2,0.2);
  \end{tikzpicture}%
  \hspace{5pt}\textcolor{MatnText}{\headfont\bfseries #1\enspace #2}\par
}

% --- Tata letak otomatis (opsional, aktif jika \WsTataLetakOtomatis didefinisikan;
%     worksheet yang tata letaknya sudah diatur manual tidak memakainya):
%     judul Level tidak tertinggal sendirian, Mini Drill & Contoh tidak terpotong,
%     kotak soal tidak dimulai di sisa halaman yang terlalu sempit ---
\ifdefined\WsTataLetakOtomatis
  \usepackage{needspace}
  \let\EkLevelA\levelA \renewcommand{\levelA}[1]{\needspace{9\baselineskip}\EkLevelA{#1}}
  \let\EkLevelB\levelB \renewcommand{\levelB}[1]{\needspace{9\baselineskip}\EkLevelB{#1}}
  \let\EkLevelC\levelC \renewcommand{\levelC}[1]{\needspace{9\baselineskip}\EkLevelC{#1}}
  \ifdefined\subtopik
    \let\EkSubtopik\subtopik \renewcommand{\subtopik}[2]{\needspace{16\baselineskip}\EkSubtopik{#1}{#2}}
  \fi
  \tcbset{
    drillbox/.append style={unbreakable},
    contohbox/.append style={unbreakable},
    contohscaffold/.append style={unbreakable},
  }
  % kotak soal tidak dimulai di sisa halaman yang terlalu sempit, kecuali worksheet
  % menandai \WsSoalFleksibel (soal boleh mulai di sisa halaman, langsung setelah contoh)
  \ifdefined\WsSoalFleksibel\else
    \tcbset{latihanbox/.append style={before={\par\needspace{8\baselineskip}\smallskip}}}
  \fi
\fi
